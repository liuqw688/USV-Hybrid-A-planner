cmake_minimum_required(VERSION 3.8)
project(usv_kinodynamic_frontend)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(hybrid_a_star_planner REQUIRED)
find_package(usv_mpc_control REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
add_library(${PROJECT_NAME} src/kinodynamic_astar.cpp)
set_target_properties(${PROJECT_NAME} PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} Eigen3::Eigen)
ament_target_dependencies(${PROJECT_NAME} nav_msgs usv_mpc_control)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
add_executable(usv_frontend_demo src/frontend_demo.cpp)
target_link_libraries(usv_frontend_demo ${PROJECT_NAME})
add_executable(usv_open_water_node src/open_water_node.cpp)
target_link_libraries(usv_open_water_node ${PROJECT_NAME})
ament_target_dependencies(usv_open_water_node tf2_ros rclcpp geometry_msgs nav_msgs std_msgs visualization_msgs)
add_executable(usv_tracking_node src/tracking_node.cpp)
target_link_libraries(usv_tracking_node ${PROJECT_NAME})
ament_target_dependencies(usv_tracking_node hybrid_a_star_planner tf2_ros tf2_geometry_msgs rclcpp geometry_msgs nav_msgs std_msgs visualization_msgs)
if(BUILD_TESTING)
  enable_testing()
  add_executable(usv_shared_force_test test/shared_force_test.cpp)
  target_link_libraries(usv_shared_force_test ${PROJECT_NAME})
  ament_target_dependencies(usv_shared_force_test hybrid_a_star_planner)
  add_test(NAME usv_frontend_shared_force COMMAND usv_shared_force_test)
  add_executable(usv_frontend_core_test test/model_search_test.cpp)
  target_link_libraries(usv_frontend_core_test ${PROJECT_NAME})
  add_test(NAME usv_frontend_model_search COMMAND usv_frontend_core_test)
  set_tests_properties(usv_frontend_model_search PROPERTIES TIMEOUT 30)
  add_executable(usv_frontend_map_test test/waterway_map_test.cpp)
  target_link_libraries(usv_frontend_map_test ${PROJECT_NAME})
  add_test(NAME usv_frontend_waterway_map COMMAND usv_frontend_map_test)
  set_tests_properties(usv_frontend_waterway_map PROPERTIES TIMEOUT 30)
endif()
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS usv_frontend_demo usv_open_water_node usv_tracking_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY docs config launch rviz DESTINATION share/${PROJECT_NAME})
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 eigen3_cmake_module nav_msgs hybrid_a_star_planner usv_mpc_control)
ament_package()
